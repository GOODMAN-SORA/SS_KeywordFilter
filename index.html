function renderData(data) { const container = document.getElementById("resultContainer"); container.innerHTML = ""; if (Object.keys(data.Characters).length === 0 && Object.keys(data.Records).length === 0) { container.innerHTML = "
검색 결과가 없습니다.
"; return; } // 1. 캐릭터 섹션 렌더링 if (Object.keys(data.Characters).length > 0) { const charSection = document.createElement("div"); charSection.className = "section"; charSection.innerHTML = "
캐릭터
"; for (const charId in data.Characters) { const char = data.Characters[charId]; const card = document.createElement("div"); card.className = "card"; let html = `
${char.Name}
`; // 스킬 렌더링 (+ 연관된 잠재력) if (Object.keys(char.skills).length > 0) { html += `
[스킬]
`; for (const skillCode in char.skills) { const skill = char.skills[skillCode]; // 스킬 이름과 태그 html += `
- ${skill.Name}
${createTagHTML(skill.tags)}
`; // 이 스킬과 연관된 잠재력이 있다면 스킬 하위에 살짝 들여쓰기해서 표시 if (skill.related_potentials && skill.related_potentials.length > 0) { html += `
`; html += `
연관 잠재력
`; skill.related_potentials.forEach(pot => { html += `
↳ ${pot.Name}
${createTagHTML(pot.tags)}
`; }); html += `
`; } } } // 공통 잠재력 렌더링 (어떤 스킬과도 매칭되지 않은 잠재력들) if (Object.keys(char.potentials).length > 0) { html += `
[공통 잠재력]
`; for (const potCode in char.potentials) { const pot = char.potentials[potCode]; html += `
- ${pot.Name}
${createTagHTML(pot.tags)}
`; } } card.innerHTML = html; charSection.appendChild(card); } container.appendChild(charSection); } // 2. 레코드 섹션 렌더링 (기존과 동일) if (Object.keys(data.Records).length > 0) { const recSection = document.createElement("div"); recSection.className = "section"; recSection.innerHTML = "
레코드
"; for (const recId in data.Records) { const rec = data.Records[recId]; const card = document.createElement("div"); card.className = "card"; let html = `
${rec.Name}
`; if (rec.main_skill.length > 0) { html += `
[메인 스킬]
${createTagHTML(rec.main_skill)}
`; } if (Object.keys(rec.sub_skills).length > 0) { for (const subType in rec.sub_skills) { html += `
[보조 스킬 ${subType}]
${createTagHTML(rec.sub_skills[subType].tags)}
`; } } card.innerHTML = html; recSection.appendChild(card); } container.appendChild(recSection); } }
